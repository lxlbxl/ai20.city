4.2 Quiz Logic & Branching
Progressive Profiling:

Adjust questions based on previous answers
Example: If "Already using AI" → Ask "Which AI tools?" instead of basic questions
If budget is low → Route to DIY/DWY services
If budget is high → Route to fCAIO/DFY services

Lead Scoring:

Assign point values to answers
High-intent signals: Immediate timeline, higher budget, specific challenges
Route hot leads differently (immediate sales follow-up vs. nurture sequence)

4.3 Quiz Technical Implementation
Technologies:

HTML forms with validation
JavaScript for step progression and branching logic
Local storage for progress saving
Form submission to backend/CRM (Webhooks)
Analytics tracking per step (Google Analytics events)
A/B testing capability for questions and copy

// Smooth Scroll for Anchor Links
function initSmoothScroll() {
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
            e.preventDefault();
            const target = document.querySelector(this.getAttribute('href'));
            if (target) {
                target.scrollIntoView({
                    behavior: 'smooth',
                    block: 'start'
                });
            }
        });
    });

// Scroll to Top Button
function initScrollToTop() {
    const scrollBtn = document.createElement('button');
    scrollBtn.innerHTML = '↑';
    scrollBtn.className = 'fixed bottom-8 right-8 bg-blue-600 text-white w-12 h-12 rounded-full shadow-lg hover:bg-blue-700 transition-all opacity-0 invisible z-40';
    scrollBtn.id = 'scroll-to-top';
    document.body.appendChild(scrollBtn);
    
    window.addEventListener('scroll', function() {
        if (window.pageYOffset > 300) {
            scrollBtn.classList.remove('opacity-0', 'invisible');
        } else {
            scrollBtn.classList.add('opacity-0', 'invisible');
        }
    });
    
    scrollBtn.addEventListener('click', function() {
        window.scrollTo({
            top: 0,
            behavior: 'smooth'
        });
    });
}

// Lazy Loading for Images
function initLazyLoading() {
    const images = document.querySelectorAll('img[data-src]');
    const imageOptions = {
        threshold: 0.1,
        rootMargin: '0px 0px 50px 0px'
    };
    
    const imageObserver = new IntersectionObserver((entries, observer) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const img = entry.target;
                img.src = img.dataset.src;
                img.removeAttribute('data-src');
                observer.unobserve(img);
            }
        });
    }, imageOptions);
    
    images.forEach(img => imageObserver.observe(img));
}

// Sticky Header on Scroll
function initStickyHeader() {
    const header = document.getElementById('main-header');
    let lastScroll = 0;
    
    window.addEventListener('scroll', function() {
        const currentScroll = window.pageYOffset;
        
        if (currentScroll > 100) {
            header.classList.add('shadow-lg');
        } else {
            header.classList.remove('shadow-lg');
        }
        
        lastScroll = currentScroll;
    });
}

            'value': value
        });
    }
}

function initFormValidation() {
    const forms = document.querySelectorAll('form[data-validate]');
    
    forms.forEach(form => {
        const inputs = form.querySelectorAll('input, select, textarea');

            // Real-time validation

                validateField(this);
            });
            
            input.addEventListener('input', function() {
                if (this.classList.contains('border-red-500')) {
                    validateField(this);
                }
            });
        });
        
        // Form submission
        form.addEventListener('submit', function(e) {
            e.preventDefault();
            
            let isValid = true;
            inputs.forEach(input => {
                if (!validateField(input)) {
                    isValid = false;
                }
            });
            
            if (isValid) {
                handleFormSubmit(form);
            }
        });
    });
}

function validateField(field) {
    let isValid = true;
    const value = field.value.trim();
    
    // Required field validation
    if (field.hasAttribute('required') && !value) {
        showFieldError(field, 'This field is required');
        isValid = false;
    }
    // Email validation
    else if (field.type === 'email' && value) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(value)) {
            showFieldError(field, 'Please enter a valid email address');
            isValid = false;
        } else {
            hideFieldError(field);
        }
    }
    // Phone validation
    else if (field.type === 'tel' && value) {
        const phoneRegex = /^[\d\s\-\+\(\)]+$/;
        if (!phoneRegex.test(value) || value.length < 10) {
            showFieldError(field, 'Please enter a valid phone number');
            isValid = false;
        } else {
            hideFieldError(field);
        }
    }
    // URL validation
    else if (field.type === 'url' && value) {
        try {
            new URL(value);
            hideFieldError(field);
        } catch {
            showFieldError(field, 'Please enter a valid URL');
            isValid = false;
        }
    }
    // Checkbox validation (for required checkboxes like consent)
    else if (field.type === 'checkbox' && field.hasAttribute('required') && !field.checked) {
        showFieldError(field, 'You must agree to continue');
        isValid = false;
    }
    else {
        hideFieldError(field);
    }
    
    return isValid;
}
